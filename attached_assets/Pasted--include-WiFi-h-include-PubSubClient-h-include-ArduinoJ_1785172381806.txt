#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <WiFiClientSecure.h>

// ===== CONFIGURACAO — PREENCHA AQUI =====
const char* WIFI_SSID   = "Autobras";
const char* WIFI_SENHA  = "Auto@2021";

const char* MQTT_HOST   = "mqtt.bluebee.ia.br";
const int   MQTT_PORTA  = 8883;
const char* MQTT_USER   = "gw-escritorio-mooca-iluminacao-6dc7bc68";
const char* MQTT_PASS   = "c01277ed5d6b24e32672367e0c53ae82";

const char* TENANT_ID   = "17656272-0f77-4481-8d77-9d756a738aff";
const char* GATEWAY_ID  = "gw-escritorio-mooca-iluminacao-6dc7bc68";
// ========================================

#define LED 2

WiFiClientSecure espClient;
PubSubClient mqtt(espClient);

String topicoTelemetria;
unsigned long ultimoEnvio = 0;

void conectarWiFi() {
  WiFi.begin(WIFI_SSID, WIFI_SENHA);
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(300);
  }
  Serial.println("\nWiFi conectado! IP: " + WiFi.localIP().toString());
}

void conectarMQTT() {
  while (!mqtt.connected()) {
    Serial.print("Conectando ao broker MQTT... ");
    String clientId = "esp32-bancada-" + String((uint32_t)ESP.getEfuseMac(), HEX);
    if (mqtt.connect(clientId.c_str(), MQTT_USER, MQTT_PASS)) {
      Serial.println("conectado!");
      digitalWrite(LED, HIGH);
    } else {
      Serial.print("falhou, rc=");
      Serial.print(mqtt.state());
      Serial.println(" — tentando de novo em 3s");
      digitalWrite(LED, LOW);
      delay(3000);
    }
  }
}

void setup() {
  pinMode(LED, OUTPUT);
  Serial.begin(115200);
  delay(1000);

  conectarWiFi();

  topicoTelemetria = String("bluebee/") + TENANT_ID +
                     "/gateway/" + GATEWAY_ID + "/sensors/bancada/esp32";
  Serial.println("Topico: " + topicoTelemetria);
 
  espClient.setInsecure(); // aceita o certificado sem validar (OK para bancada)
  mqtt.setServer(MQTT_HOST, MQTT_PORTA);
  conectarMQTT();
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) conectarWiFi();
  if (!mqtt.connected()) conectarMQTT();
  mqtt.loop();

  if (millis() - ultimoEnvio >= 10000) {
    ultimoEnvio = millis();

    // valores simulados que variam suavemente
    float temp    = 24.0 + 3.0 * sin(millis() / 60000.0);
    float umidade = 55.0 + 8.0 * sin(millis() / 90000.0 + 1.5);

    StaticJsonDocument<128> doc;
    doc["temp"]    = round(temp * 10) / 10.0;
    doc["umidade"] = round(umidade * 10) / 10.0;
    doc["rele"]    = false;

    char payload[128];
    serializeJson(doc, payload);

    bool ok = mqtt.publish(topicoTelemetria.c_str(), payload);
    Serial.printf("Publicado (%s): %s\n", ok ? "OK" : "FALHOU", payload);
  }
}

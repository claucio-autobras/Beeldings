# Tarefa: tornar a escrita BACnet do agente resiliente à perda do SimpleACK (cold device)
# Arquivo: infra/gateway-agent/index.mjs — NÃO mexer em apps/.

Bug: a 1ª WriteProperty a uma controladora "fria" perde o SimpleACK no ARP/binding inicial.
node-bacnet NÃO retransmite, então o callback do writeProperty NUNCA dispara — a saída ATUA, mas o
agente nunca publica commands/result e o backend cai em timeout. Hoje a escrita faz UM writeProperty,
SEM timeout próprio e SEM retry. Corrigir com (1) timeout POR TENTATIVA e (2) retransmissão idempotente.

## 1) Constantes (junto das demais constantes BACnet)
  const WRITE_APDU_TIMEOUT_MS = Number(process.env.BACNET_WRITE_TIMEOUT_MS ?? 4000); // por tentativa
  const WRITE_MAX_ATTEMPTS    = Math.max(1, Number(process.env.BACNET_WRITE_RETRIES ?? 2));

## 2) bacnetWriteProperty — adicionar timeout por tentativa (REJEITA se o ACK não vier)
  function bacnetWriteProperty(address, objectId, property, valueList, priority, timeoutMs = WRITE_APDU_TIMEOUT_MS) {
    return new Promise((resolve, reject) => {
      let settled = false;
      const timer = setTimeout(() => {
        if (settled) return;
        settled = true;
        reject(new Error(`confirmação BACnet não recebida em ${timeoutMs}ms`));
      }, timeoutMs);
      bacnetClient.writeProperty(address, objectId, property, valueList, { priority }, (err) => {
        if (settled) return;
        settled = true;
        clearTimeout(timer);
        err ? reject(err) : resolve();
      });
    });
  }

## 3) writeBacnet — envolver a escrita num loop de retransmissão
  const applicationTag = ANALOG_OBJECT_TYPES.has(objectType) ? BACNET_TAG_REAL : BACNET_TAG_ENUMERATED;
  let lastError = 'falha desconhecida';
  for (let attempt = 1; attempt <= WRITE_MAX_ATTEMPTS; attempt++) {
    try {
      await bacnetWriteProperty(ip, { type: objectType, instance: objectInstance },
        PRESENT_VALUE, [{ type: applicationTag, value }], priority);
      const suffix = attempt > 1 ? ` (tentativa ${attempt})` : '';
      console.log(`✓ write OK${suffix} — ${ip} obj=${objectType}:${objectInstance} value=${value} prio=${priority}`);
      return { success: true };
    } catch (err) {
      lastError = err instanceof Error ? err.message : String(err);
      console.error(`✗ write tentativa ${attempt}/${WRITE_MAX_ATTEMPTS} FALHOU — ${ip} obj=${objectType}:${objectInstance}: ${lastError}`);
    }
  }
  return { success: false, error: lastError };

## 4) TRAVA DE TEMPO (conferir no backend — crítico)
  O orçamento do agente (WRITE_MAX_ATTEMPTS × WRITE_APDU_TIMEOUT_MS ≈ 8s + ida/volta MQTT) tem que
  ser MENOR que o WRITE_TIMEOUT_MS do backend em apps/backend/.../bacnet-write.service.ts.
  Garantir WRITE_TIMEOUT_MS ≈ 15000ms. Se estiver menor que ~8s, o usuário continua vendo ERRO mesmo
  com a 2ª tentativa recuperando o ACK.

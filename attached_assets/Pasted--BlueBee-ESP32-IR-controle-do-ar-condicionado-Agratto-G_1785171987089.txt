/**
 * BlueBee — ESP32 IR (controle do ar-condicionado Agratto/Gree)
 * =============================================================
 * Atua como "controle remoto MQTT": recebe comandos do BlueBee e
 * dispara sinais infravermelho no protocolo Gree (Agratto = OEM Gree).
 *
 * Telemetria (estado PRESUMIDO — IR não tem retorno do aparelho):
 *   Tópico:  bluebee/{TENANT_ID}/gateway/{GATEWAY_ID}/sensors/bancada/esp32-ir
 *   Payload: {"ligado": false, "setpoint": 23}
 *
 * Comando:
 *   Subscreve: .../sensors/bancada/esp32-ir/cmd
 *   Payloads:  {"id": 123, "tag": "ligado",   "value": 1}
 *              {"id": 124, "tag": "setpoint", "value": 22}
 *   Resposta:  .../sensors/bancada/esp32-ir/cmd/rpc → {"id": 123, "value": true}
 *
 * Placa: "ESP32 Dev Module".
 */

#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <IRremoteESP8266.h>
#include <ir_Gree.h>

#include <IRsend.h>


// ========================== CONFIGURAÇÃO ==========================
#define WIFI_SSID     "Autobras"
#define WIFI_PASS     "Auto@2021"

#define MQTT_HOST     "mqtt.bluebee.ia.br"
#define MQTT_PORT     8883
#define MQTT_USER     "gw-escritorio-mooca-iluminacao-6dc7bc68"      // mesmo usuário *-sensors da bancada
#define MQTT_PASS     "c01277ed5d6b24e32672367e0c53ae82"
#define TENANT_ID     "17656272-0f77-4481-8d77-9d756a738aff"
#define GATEWAY_ID    "gw-escritorio-mooca-iluminacao-6dc7bc68"

#define TELEMETRY_INTERVAL_MS 10000UL

// --- IR ---
#define PIN_IR_LED    4       // LED IR: GPIO4 -> resistor ~220R -> LED -> GND
// Se o ar NÃO responder, teste variantes do modelo Gree (algumas unidades
// precisam): ac.setModel(gree_ac_remote_model_t::YAW1F);  // ou YBOFB
// ==================================================================

const String TOPIC_BASE      = String("bluebee/") + TENANT_ID + "/gateway/" + GATEWAY_ID + "/sensors/bancada/esp32-ir";
const String TOPIC_TELEMETRY = TOPIC_BASE;
const String TOPIC_CMD       = TOPIC_BASE + "/cmd";
const String TOPIC_CMD_RPC   = TOPIC_BASE + "/cmd/rpc";

WiFiClientSecure tls;
PubSubClient mqtt(tls);
IRGreeAC ac(PIN_IR_LED);

// novo da TV
IRsend irsend(PIN_IR_LED);   // mesmo LED, usado para as TVs

// Power toggle das TVs (mesmo código liga e desliga)
#define TV_SAMSUNG_POWER 0xE0E040BF
#define TV_LG_POWER      0x20DF10D0

// Estado PRESUMIDO (último comando enviado — IR é "só ida")
bool    acOn      = false;
uint8_t acSetTemp = 23;      // 16..30 °C

unsigned long lastTelemetry = 0;

void sendIr() {
  if (acOn) {
    ac.on();
    ac.setMode(kGreeCool);
    ac.setTemp(acSetTemp);
    ac.setFan(kGreeFanAuto);
  } else {
    ac.off();
  }
  ac.send();
  Serial.printf("[IR] enviado: ligado=%d setpoint=%d\n", acOn, acSetTemp);
}

void publishTelemetry() {
  JsonDocument doc;
  doc["ligado"]   = acOn;
  doc["setpoint"] = acSetTemp;

  char buf[128];
  size_t n = serializeJson(doc, buf, sizeof(buf));
  mqtt.publish(TOPIC_TELEMETRY.c_str(), (const uint8_t*)buf, n, false);
  Serial.printf("[telemetria] %s\n", buf);
}

void onMqttMessage(char* topic, byte* payload, unsigned int length) {
  if (TOPIC_CMD != topic) return;

  JsonDocument doc;
  DeserializationError err = deserializeJson(doc, payload, length);
  if (err) {
    Serial.printf("[cmd] payload inválido: %s\n", err.c_str());
    return;
  }

  const char* tag = doc["tag"] | "ligado";   // sem tag = assume ligado
  JsonVariant v = doc["value"];

  JsonDocument resp;
  if (!doc["id"].isNull()) resp["id"] = doc["id"];

  if (strcmp(tag, "setpoint") == 0) {
    int t = v.as<int>();
    if (t < 16) t = 16;
    if (t > 30) t = 30;
    acSetTemp = (uint8_t)t;
    // Ajustar temperatura com o ar desligado não liga o ar; só envia IR se ligado
    if (acOn) sendIr();
    resp["value"] = acSetTemp;
    Serial.printf("[cmd] setpoint -> %d\n", acSetTemp);
  } else {  // "ligado"
    bool on = v.is<bool>() ? v.as<bool>() : (v.as<float>() != 0);
    acOn = on;
    sendIr();
    resp["value"] = acOn;
    Serial.printf("[cmd] ligado -> %s\n", acOn ? "ON" : "OFF");
  } 

  // novo TV
    if (strcmp(tag, "setpoint") == 0) {
    int t = v.as<int>();
    if (t < 16) t = 16;
    if (t > 30) t = 30;
    acSetTemp = (uint8_t)t;
    if (acOn) sendIr();
    resp["value"] = acSetTemp;
    Serial.printf("[cmd] setpoint -> %d\n", acSetTemp);
  } else if (strcmp(tag, "tv_samsung") == 0) {
    irsend.sendSAMSUNG(TV_SAMSUNG_POWER);
    resp["value"] = true;
    Serial.println("[IR] TV Samsung: power toggle enviado");
  } else if (strcmp(tag, "tv_lg") == 0) {
    irsend.sendNEC(TV_LG_POWER);
    resp["value"] = true;
    Serial.println("[IR] TV LG: power toggle enviado");
  } else {  // "ligado"
    bool on = v.is<bool>() ? v.as<bool>() : (v.as<float>() != 0);
    acOn = on;
    sendIr();
    resp["value"] = acOn;
    Serial.printf("[cmd] ligado -> %s\n", acOn ? "ON" : "OFF");
  }

  char buf[128];
  size_t n = serializeJson(resp, buf, sizeof(buf));
  mqtt.publish(TOPIC_CMD_RPC.c_str(), (const uint8_t*)buf, n, false);

  publishTelemetry();
  lastTelemetry = millis();
}

void connectWifi() {
  if (WiFi.status() == WL_CONNECTED) return;
  Serial.printf("Wi-Fi: conectando em %s", WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.printf("\nWi-Fi OK, IP %s\n", WiFi.localIP().toString().c_str());
}

void connectMqtt() {
  while (!mqtt.connected()) {
    connectWifi();
    String clientId = String("esp32-ir-") + String((uint32_t)ESP.getEfuseMac(), HEX);
    Serial.printf("MQTT: conectando em %s:%d... ", MQTT_HOST, MQTT_PORT);
    if (mqtt.connect(clientId.c_str(), MQTT_USER, MQTT_PASS)) {
      Serial.println("OK");
      mqtt.subscribe(TOPIC_CMD.c_str(), 1);
      Serial.printf("Subscrito em %s\n", TOPIC_CMD.c_str());
    } else {
      Serial.printf("falhou (rc=%d), tentando em 3s\n", mqtt.state());
      delay(3000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(200);
  Serial.println("\nBlueBee ESP32 IR — ar-condicionado Agratto (protocolo Gree)");

  ac.begin();
  //novo TV 
  irsend.begin();

  ac.setMode(kGreeCool);
  ac.setTemp(acSetTemp);
  ac.setFan(kGreeFanAuto);
  ac.setModel(gree_ac_remote_model_t::YBOFB);  // descomente se não responder

  tls.setInsecure();  // trocar por setCACert(...) quando fixarmos a cadeia
  mqtt.setServer(MQTT_HOST, MQTT_PORT);
  mqtt.setCallback(onMqttMessage);
  mqtt.setBufferSize(512);

  connectMqtt();
  publishTelemetry();
  lastTelemetry = millis();
}

void loop() {
  if (!mqtt.connected()) connectMqtt();
  mqtt.loop();

  if (millis() - lastTelemetry >= TELEMETRY_INTERVAL_MS) {
    publishTelemetry();
    lastTelemetry = millis();
  }
}
